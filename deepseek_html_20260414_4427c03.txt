    function attachHandlers() {
        const nameInput = document.getElementById('g_name');
        const phoneInput = document.getElementById('g_phone');
        const dateInput = document.getElementById('g_date');
        const timeInput = document.getElementById('g_time');
        const commentInput = document.getElementById('g_comment');
        const submitBtn = document.getElementById('submitBtn');

        // Обработка ошибок и валидации
        function validateForm() {
            let isValid = true;
            hideErrors();

            const name = nameInput.value.trim();
            if (name.length < 2) {
                showError('error_name');
                nameInput.style.borderColor = 'var(--vkui--color_stroke_negative)';
                isValid = false;
            }

            const phone = phoneInput.value.trim();
            const digits = phone.replace(/\D/g, '');
            if (digits.length < 10) {
                showError('error_phone');
                phoneInput.style.borderColor = 'var(--vkui--color_stroke_negative)';
                isValid = false;
            }

            const date = dateInput.value;
            if (!date) {
                showError('error_date');
                dateInput.style.borderColor = 'var(--vkui--color_stroke_negative)';
                isValid = false;
            } else {
                const selected = new Date(date);
                const today = new Date();
                today.setHours(0,0,0,0);
                if (selected < today) {
                    const errDiv = document.getElementById('error_date');
                    errDiv.innerHTML = '⚠️ Дата не может быть в прошлом';
                    showError('error_date');
                    dateInput.style.borderColor = 'var(--vkui--color_stroke_negative)';
                    isValid = false;
                }
            }

            if (!timeInput.value) {
                showError('error_time');
                timeInput.style.borderColor = 'var(--vkui--color_stroke_negative)';
                isValid = false;
            }

            return isValid;
        }

        function hideErrors() {
            ['error_name','error_phone','error_date','error_time'].forEach(id => {
                const el = document.getElementById(id);
                if (el) el.style.display = 'none';
            });
            [nameInput, phoneInput, dateInput, timeInput].forEach(inp => {
                if (inp) inp.style.borderColor = '';
            });
        }

        function showError(id) {
            const el = document.getElementById(id);
            if (el) el.style.display = 'flex';
        }

        async function sendGroom() {
            if (!validateForm()) return;

            const btn = submitBtn;
            const originalText = btn.innerHTML;
            btn.innerHTML = '<span class="loader"></span> Отправка...';
            btn.disabled = true;

            const now = new Date();
            const order = {
                name: nameInput.value.trim(),
                phone: phoneInput.value.trim(),
                date: dateInput.value,
                time: timeInput.value,
                comment: commentInput.value.trim(),
                createdDate: now.toLocaleDateString('ru-RU'),
                createdTime: now.toLocaleTimeString('ru-RU'),
                status: 'Новая',
                source: 'VK Mini App',
                vk_user_id: vkUser.id || null,
                timestamp: firebase.database.ServerValue.TIMESTAMP
            };

            try {
                await db.ref('orders').push(order);
                // Показать уведомление через VK Bridge
                if (vkBridge) {
                    vkBridge.send('VKWebAppShowNotification', {
                        text: '✅ Заявка отправлена! Мы свяжемся с вами в ближайшее время 🐾',
                        buttons: [{ text: 'OK' }]
                    });
                }
                // Тост-сообщение
                showToast('Заявка успешно отправлена! Спасибо 😊');
                // Очистить форму
                nameInput.value = '';
                phoneInput.value = '';
                dateInput.value = '';
                timeInput.value = '';
                commentInput.value = '';
                hideErrors();
                setMinDate();
            } catch (err) {
                console.error(err);
                showToast('Ошибка при отправке. Попробуйте позже.', true);
            } finally {
                btn.innerHTML = originalText;
                btn.disabled = false;
            }
        }

        function showToast(message, isError = false) {
            const toast = document.createElement('div');
            toast.className = 'success-toast';
            toast.style.background = isError ? 'var(--vkui--color_background_negative)' : 'var(--vkui--color_background_positive)';
            toast.innerText = message;
            document.body.appendChild(toast);
            setTimeout(() => toast.remove(), 3000);
        }

        function setMinDate() {
            const today = new Date();
            const yyyy = today.getFullYear();
            const mm = String(today.getMonth() + 1).padStart(2,'0');
            const dd = String(today.getDate()).padStart(2,'0');
            dateInput.min = `${yyyy}-${mm}-${dd}`;
        }

        submitBtn.onclick = sendGroom;

        // Живая валидация
        [nameInput, phoneInput, dateInput, timeInput].forEach(inp => {
            inp.addEventListener('input', () => {
                inp.style.borderColor = '';
                const errId = `error_${inp.id.replace('g_','')}`;
                const errDiv = document.getElementById(errId);
                if (errDiv) errDiv.style.display = 'none';
            });
        });

        // Маска телефона
        phoneInput.addEventListener('input', function(e) {
            let value = this.value.replace(/\D/g, '');
            if (value.length > 11) value = value.slice(0,11);
            let formatted = '';
            if (value.length > 0) {
                formatted = '+7';
                if (value[0] === '7') value = value.substring(1);
                if (value.length > 0) formatted += ' (' + value.substring(0,3);
                if (value.length >= 4) formatted += ') ' + value.substring(3,6);
                if (value.length >= 7) formatted += '-' + value.substring(6,8);
                if (value.length >= 9) formatted += '-' + value.substring(8,10);
                this.value = formatted;
            } else {
                this.value = '';
            }
        });
    }
